(defgroup my/frame-transparency nil
  "Cross-platform frame transparency."
  :group 'frames)

(defcustom my-opacity 86
  "Background opacity for graphical Emacs frames.
100 means opaque, 0 means fully transparent."
  :type 'integer
  :group 'my/frame-transparency)

(defcustom my-macos-blur-radius 10
  "Blur radius for patched macOS NS Emacs builds."
  :type 'integer
  :group 'my/frame-transparency)

(defun my/macos-ns-p (&optional frame)
  "Non-nil when FRAME is a macOS NS graphical frame."
  (and (featurep 'ns)
       (display-graphic-p (frame-parameter frame 'display))
       (eq (framep frame) 'ns)))

(defun my/linux-gui-p (&optional frame)
  "Non-nil when FRAME is a Linux graphical frame."
  (and (display-graphic-p (frame-parameter frame 'display))
       (memq (framep frame) '(x pgtk))))

(defun my/frame-transparency-parameters (&optional frame)
  "Return transparency parameters suitable for FRAME."
  (cond
   ;; macOS NS build with bbenchen/homebrew-emacs-plus blur patch.
   ((my/macos-ns-p frame)
    `((alpha-background . ,my-opacity)
      (ns-background-blur . ,my-macos-blur-radius)
      (ns-alpha-elements . (ns-alpha-all))))

   ;; Linux X11 / PGTK.
   ;; Blur itself is compositor-side: picom, Hyprland, KDE, etc.
   ((my/linux-gui-p frame)
    `((alpha-background . ,my-opacity)))

   ;; Terminal or unsupported frame.
   (t nil)))

(defun my/apply-frame-transparency (&optional frame)
  "Apply transparency to FRAME."
  (let ((frame (or frame (selected-frame))))
    (when-let* ((params (my/frame-transparency-parameters frame)))
      (modify-frame-parameters frame params))))

(defun my/add-default-frame-transparency-parameters ()
  "Add transparency defaults for future frames.

Important for daemon/emacsclient: some parameters, especially
`ns-background-blur', need to be present at frame creation time."
  ;; Portable Emacs 29+ background-only transparency.
  (add-to-list 'default-frame-alist `(alpha-background . ,my-opacity))

  ;; Only add NS-specific parameters in an NS-capable macOS build.
  ;; In daemon mode, `window-system' can be nil, but `(featurep 'ns)'
  ;; is still true in the macOS NS build.
  (when (featurep 'ns)
    (add-to-list 'default-frame-alist
                 `(ns-background-blur . ,my-macos-blur-radius))
    (add-to-list 'default-frame-alist
                 '(ns-alpha-elements . (ns-alpha-all)))))

(my/add-default-frame-transparency-parameters)

(unless (daemonp)
  (add-hook 'window-setup-hook #'my/apply-frame-transparency))

(when (display-graphic-p)
  (my/apply-frame-transparency))

    (defun my/rebuild-persp-order (&optional current-persps base-order)
      "Return perspective order merged with CURRENT-PERPS.
Current order is preserved for existing perspectives and new ones are appended."
      (let* ((current-persps (or current-persps
                                 (remove (my/persp-none-name) persp-names-cache)))
             (base-order (or base-order my-dynamic-persps))
             (ordered-existing
              (seq-filter (lambda (name) (member name current-persps))
                          base-order))
             (new-persps
              (seq-remove (lambda (name) (member name ordered-existing))
                          current-persps)))
        (append ordered-existing new-persps)))

             (let ((name (and (fboundp 'safe-persp-name)
                              (safe-persp-name persp))))
               (when (and name (not (string= name (my/persp-none-name))))
                 (push name names))))
           *persp-hash*))
        (nreverse names)))

    (defun my/apply-persp-order ()
      "Apply the perspective order described in the current reorder buffer."
      (interactive)
      (unless (string= (buffer-name) my-persp-order-buffer-name)
        (user-error "Not in the perspective order buffer"))
      (let* ((current-persps (my/persp-current-names))
             (new-order (my/persp-order-buffer-contents))
             (unknown (seq-remove (lambda (name) (member name current-persps)) new-order))
             (missing (seq-remove (lambda (name) (member name new-order)) current-persps))
             (duplicates (seq-filter (lambda (name) (> (cl-count name new-order :test #'string=) 1))
                                     new-order)))
        (cond
         (unknown
          (user-error "Unknown perspectives in buffer: %s" (string-join (delete-dups unknown) ", ")))
         (missing
          (user-error "Missing perspectives from buffer: %s" (string-join missing ", ")))
         (duplicates
          (user-error "Duplicate perspectives in buffer: %s" (string-join (delete-dups duplicates) ", ")))
         (t
          (my/apply-persp-order-to-runtime new-order t)
          (quit-restore-window (selected-window) 'kill)
          (message "Perspective order updated: %s" (string-join my-dynamic-persps ", "))))))

(defun ox/minibuffer-backward-kill (arg)
  "When minibuffer is completing a file name delete up to parent
folder, otherwise delete a word"
  (interactive "p")
  (if minibuffer-completing-file-name
      ;; Borrowed from https://github.com/raxod502/selectrum/issues/498#issuecomment-803283608
      (if (string-match-p "/." (minibuffer-contents))
	  (zap-up-to-char (- arg) ?/)
	(delete-minibuffer-contents))
    (backward-kill-word arg)))

  (if (zerop vertico--total)
      (vertico-exit-input)
    (let* ((candidate-index (max 0 vertico--index))
           (candidate (let ((vertico--index candidate-index))
                        (substring-no-properties (vertico--candidate)))))

** vterm
using (setq term-next-prompt "prompt") is not recommanded it does follow empty lines
the correct way of doing it is to create our own prompt regexp system see: [[(defun my/vterm-next-prompt][my-vterm-next-prompt]]
#+begin_src emacs-lisp
(use-package vterm
    :straight t
    :defer 0
    :after (general which-key)
    :config
    ;; TRAMP's control shell gives LC_CTYPE a literal pair of quote characters
    ;; to keep its protocol locale-neutral.  A remote VTerm otherwise passes it to
    ;; the interactive shell, which makes Bash emit a setlocale warning.
    ;; A name-only process environment entry unsets the variable; the remote
    ;; shell then uses its valid LANG/LC_ALL locale without host-specific paths
    ;; or locale discovery.
    (add-to-list 'vterm-environment "LC_CTYPE")

    ;; Make C-h deleting correctly backward
    ;; Without it either start help in Emacs mode
    ;; Or erroring for buffer in read-only in Evil mode
    (define-key vterm-mode-map (kbd "C-h") #'vterm-send-backspace)
    (define-key vterm-mode-map (kbd "DEL") #'vterm-send-backspace)
    (with-eval-after-load 'evil
        ;; Important: bind in Evil's insert state for vterm
        (evil-define-key '(insert) vterm-mode-map

* My-switch-to-persp-vterm-by-number
Creating a function to target a specific vterm buffer inside a specific perspective.
It first parse every buffers in the perspective seeking for vterm buffers only in the good order.
Then it switch to the vterm buffer by it's number.

the loop bind keys to this function.
The current-i variable is a workaround to prevent elisp dynamical scope in the lambda to only catch the i reference and having the correct number to bind by taking the good i value at each iteration of the loop.

As none have every buffers we just switch to the vterm buffer by it's number.

Unbind "*C-\*", then remap toggle-input-method to "*C-|*" so we can allow "*C-\*" as leader key to travel between vterms

  	 (persp-vterm-buffers (cl-remove-if-not (lambda (buf) (string-match-p "^\\*vterminal<[0-9]+>\\*$" (buffer-name buf))) all-buffers-in-persp)))

  	(if (get-current-persp)

  	      (while (< index number)
  		(setq index (+ 1 index)))

        )

  "Make and Run a C program on a vterm buffer based on the makefile recipies
because compile mode is too slow"
  (interactive)
  (if (or
	  (eq major-mode 'c-ts-mode)

	(save-buffer)

	  (unless (not (eq switched nil))

	  (vterm-send-string target)))

(defun ox/org-buffer-property (name)
  "Get the value of a buffer-wide Org property NAME."
  (save-excursion
    (goto-char (point-min))
    (if (re-search-forward (concat "^#\\+PROPERTY:.*" name " +\\(.*\\)") nil t)
        (progn
          (message "Found property %s with value %s" name (match-string 1))
          (match-string 1))
      (message "Property %s not found" name)
      nil)))

  "Automatically tangle Org files with the :auto-tangle property set to t."
  (message "Running ox/org-babel-tangle-config")
  (let ((org-confirm-babel-evaluate nil))
    (org-babel-tangle)))

(defun ox/check-and-add-tangle-hook ()
  "Check for the auto-tangle property and add tangle hook if needed."
  (message "Checking for auto-tangle property...")
  (let ((auto-tangle (ox/org-buffer-property ":auto-tangle")))
    (if (and auto-tangle (string= auto-tangle "t"))
        (progn
          (message "Auto-tangle property found. Adding after-save-hook...")
          (add-hook 'after-save-hook #'ox/org-babel-tangle-config nil 'local))
      (message "Auto-tangle property not set to t"))))

  "Return non-nil when a straight update prompt is due."
  (let ((last-run (ox/straight-update-last-run-time)))
    (or (null last-run)
        (>= (float-time (time-subtract (current-time) last-run))
            (* ox/straight-update-interval-days 24 60 60)))))

  (expand-file-name "straight/repos/" straight-base-dir))

(defun ox/straight--git-lines (repo &rest args)
  "Run git ARGS in REPO and return output lines."
  (let ((default-directory repo))
    (with-temp-buffer
      (let ((exit-code (apply #'process-file "git" nil t nil args)))
        (unless (zerop exit-code)
          (error "git %s failed in %s:\n%s"
                 (string-join args " ")
                 repo
                 (buffer-string)))
        (split-string (buffer-string) "\n" t)))))

  (ox/straight--git-lines repo "status" "--porcelain=v1" "--untracked-files=normal"))

(defun ox/straight-dirty-repos ()
  "Return an alist of dirty straight repos.

Each element is of the form:

  (REPO-NAME . STATUS-LINES)"
  (let ((repos-dir (ox/straight--repos-dir)))
    (when (file-directory-p repos-dir)
      (cl-loop for file in (directory-files repos-dir t directory-files-no-dot-files-regexp)
               when (and (file-directory-p file)
                         (file-exists-p (expand-file-name ".git" file)))
               for status = (ox/straight--repo-dirty-p file)
               when status
               collect (cons (file-name-nondirectory file) status)))))

(defun ox/straight--recipe-package-alist ()
  "Return an alist mapping straight local repo names to package symbols."
  (unless (boundp 'straight--recipe-cache)
    (error "straight--recipe-cache is not available; straight.el is not loaded yet"))
  (let (result)
    (maphash
     (lambda (package recipe)
       (let ((repo (plist-get recipe :local-repo)))
         (when repo
           (push (cons repo package) result))))
     straight--recipe-cache)
    result))

(defun ox/straight--clean-packages (dirty-repos)
  "Return straight package symbols whose repos are not in DIRTY-REPOS."
  (let ((dirty-names (mapcar #'car dirty-repos))
        (repo->package (ox/straight--recipe-package-alist))
        result)
    (dolist (entry repo->package)
      (pcase-let ((`(,repo . ,package) entry))
        (unless (member repo dirty-names)
          (cl-pushnew package result))))
    (nreverse result)))

(defun ox/straight-report-dirty-repos (&optional dirty-repos)
  "Display DIRTY-REPOS in a compact buffer."
  (interactive)
  (let ((dirty-repos (or dirty-repos (ox/straight-dirty-repos))))
    (if (null dirty-repos)
        (message "No dirty straight repos.")
      (with-current-buffer (get-buffer-create "*straight dirty repos*")
        (let ((inhibit-read-only t))
          (erase-buffer)
          (insert "Dirty straight.el repos:\n\n")
          (dolist (entry dirty-repos)
            (pcase-let ((`(,repo . ,lines) entry))
              (insert (format "%s\n" repo))
              (dolist (line lines)
                (insert (format "  %s\n" line)))
              (insert "\n")))
          (goto-char (point-min))
          (view-mode 1))
        (display-buffer (current-buffer))))))

(defun ox/straight-update-clean-packages ()
  "Pull clean straight packages, skipping dirty repos."
  (let* ((dirty-repos (ox/straight-dirty-repos))
         (clean-packages (ox/straight--clean-packages dirty-repos))
         (failures 0))
    (when dirty-repos
      (ox/straight-report-dirty-repos dirty-repos)
      (message "Skipping %d dirty straight repos: %s"
               (length dirty-repos)
               (string-join (mapcar #'car dirty-repos) ", ")))
    (dolist (package clean-packages)
      (condition-case err
          (straight-pull-package package)
        (error
         (cl-incf failures)
         (message "Failed to pull %S: %s" package (error-message-string err)))))
    (straight-rebuild-all)
    (if (zerop failures)
        (progn
          (ox/straight-update-record-run)
          (message "Straight update finished. Updated clean packages; skipped dirty repos."))
      (message "Straight update finished with %d pull failure(s); last-run timestamp not updated."
               failures))))

(defun ox/straight-update-all ()
  "Update straight packages without involving package.el."
  (interactive)
  (if ox/straight-update-skip-dirty-repos
      (ox/straight-update-clean-packages)
    (straight-pull-all)
    (straight-rebuild-all)
    (ox/straight-update-record-run)))

    (when (or (not ox/straight-update-prompt-before-update)
              (y-or-n-p "Update straight packages now? "))
      (ox/straight-update-all))))

(defun leji/acp-mobile--caffeinate-sentinel (process event)
  "Report when the ACP mobile caffeinate PROCESS finishes with EVENT."
  (when (memq (process-status process) '(exit signal))
    (when (eq process leji/acp-mobile-caffeinate-process)
      (setq leji/acp-mobile-caffeinate-process nil))

(defun leji/acp-mobile--stop-caffeinate ()
  "Stop the tracked ACP mobile caffeinate process.
Return non-nil when a live process was stopped."
  (when (process-live-p leji/acp-mobile-caffeinate-process)
    (let ((process leji/acp-mobile-caffeinate-process))
      (setq leji/acp-mobile-caffeinate-process nil)
      (set-process-sentinel process #'ignore)
      (delete-process process)
      t)))

(defun leji/acp-mobile-keep-awake (&optional duration unit)
  "Toggle whether this Mac is kept reachable for ACP mobile.
When inactive, keep the Mac awake for one hour by default.  With a prefix
argument, prompt for DURATION and UNIT, which must be either `minutes' or
`hours'.  Programmatic callers may pass DURATION and UNIT directly.  When
already active, stop immediately and restore normal macOS sleep behavior."
  (interactive
   (progn
     (unless (eq system-type 'darwin)
       (user-error "ACP mobile keep-awake is available only on macOS"))
     (if (or (process-live-p leji/acp-mobile-caffeinate-process)
             (not current-prefix-arg))
         (list nil nil)
       (list
        (read-number "Keep Mac awake for: " 1)
        (intern
         (completing-read
          "Unit: " '("minutes" "hours") nil t nil nil "hours"))))))
  (unless (eq system-type 'darwin)
    (user-error "ACP mobile keep-awake is available only on macOS"))
  (if (leji/acp-mobile--stop-caffeinate)
      (message "ACP mobile keep-awake disabled; normal macOS sleep restored")
    (setq duration (or duration 1)
          unit (or unit 'hours))
    (unless (> duration 0)
      (user-error "Duration must be greater than zero"))
    (let* ((multiplier
            (pcase unit
              ('minutes 60)
              ('hours 3600)
              (_ (user-error "Unit must be minutes or hours"))))
           (seconds (ceiling (* duration multiplier))))
      (setq leji/acp-mobile-caffeinate-process
            (make-process
             :name "acp-mobile-caffeinate"
             :command
             (list "/usr/bin/caffeinate" "-dimsu" "-t"
                   (number-to-string seconds))
             :noquery t
             :sentinel #'leji/acp-mobile--caffeinate-sentinel))
      (message "Keeping this Mac reachable for ACP mobile for %g %s"
               duration unit))))

* direnv
this package provides [[https://direnv.net/][direnv]] integration for emacs.

it works by invoking direnv to obtain the environment for the current file, then updating the emacs variables process-environment and exec-path.

the result is that programs started from within emacs, such as inferior shells, linters, compilers, and test runners, will be looked up in the correct $PATH, and will be started with the correct environment variables set.

to get started, add this to your init.el:
#+begin_src emacs-lisp
(use-package direnv
    :straight t
    :config
    (direnv-mode))
#+end_src
